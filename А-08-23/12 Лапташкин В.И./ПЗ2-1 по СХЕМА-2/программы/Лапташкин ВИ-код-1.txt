; Лапташкин В.И., А-08-23, 12, v1
; ---- vars.inc (вклеен mpscode) ----
; Параметры варианта А-08, M=12 (ТЗ 2026). Сгенерировано mpsgen — не править руками.

VAR_M        EQU 12            ; номер варианта
VAR_G        EQU 8            ; номер группы
Y2_CONST     EQU 20            ; G+M: Y2 = (Y2_CONST + X1 + X2) mod 256

; --- адреса внешних устройств (movx) ---
ADR_IND      EQU 0A000h        ; индикатор, CS Y5
ADR_KB       EQU 0C000h        ; клавиатура: запись столбцов / чтение строк, CS Y6
ADR_Y2       EQU 8000h        ; регистр Y2, CS Y4
BUF_START    EQU 6000h        ; первая ячейка буфера, CS Y3
BUF_END      EQU 65F8h        ; первая ячейка ЗА буфером (точка заворота)
BUF_SIZE     EQU 1528

; --- клавиатура 4x3 и индикатор ---
KB_COLS      EQU 4             ; столбцы Col1…Col4 — биты D0…D3 при записи в ADR_KB (0 — столбец опрашивается)
KB_ROWS      EQU 3             ; строки Row1…Row3 — биты D0…D2 при чтении ADR_KB (0 — замкнута); старшие биты не определены — маскировать
IND_OFF      EQU 0FFh           ; индикатор погашен (общий анод: 0 зажигает сегмент; D0…D7 = a…g, dp)

; --- внутреннее ОЗУ ---
HEAD_L       EQU 40h          ; «голова», младший байт
HEAD_H       EQU 41h          ; «голова», старший байт
TAIL_L       EQU 42h          ; «хвост», младший байт
TAIL_H       EQU 43h          ; «хвост», старший байт
F_EMPTY      BIT 0Ch          ; флаг пустоты буфера (21h.4)
F_OVF        BIT 14h          ; флаг переполнения буфера (22h.4)

; --- выводы ---
PIN_Y1       BIT P1.5         ; строб Y1, активный 0
PIN_Y2       BIT P1.6         ; строб Y2, активный 0
PIN_CSEN     BIT P3.4         ; CS_EN: разрешение дешифратора CS (1 — CS активны)

; --- таймеры (кварц 12 МГц) ---
T2_RELOAD_H  EQU 0F7h          ; строб Y2 2068 мкс, Timer1
T2_RELOAD_L  EQU 0ECh
TICK_H       EQU 0F8h          ; тик 2 мс для Y1/T3, Timer0
TICK_L       EQU 30h
Y1_TICKS     EQU 149           ; T1 = 298 мс
T3_TICKS     EQU 1410          ; T3 = 2820 мс
; ---- конец vars.inc ----
DSEG AT 30h
; переменные в ОЗУ не нужны: всё держим в регистрах R2-R6 и сохраняем в стеке
CSEG AT 0h
        sjmp START
org 03h                         ; INT0 — нажатие клавиши: процедура вызывается отсюда
        call KbScan ; %proc%
        reti
org 0bh ; "заглушка" для Tmr0_ovf
        nop
        reti
org 13h ; "заглушка" для INT1
        nop
        reti
org 1bh ; "заглушка" для Tmr1_ovf
        nop
        reti
org 23h ; "заглушка" для UART
        nop
        reti
org 2bh
START:
; Инициализация МК **********************************************
        mov SP, #07h            ; стек сразу за банком 0
        setb PIN_CSEN           ; разрешить дешифратор CS (без него movx никуда не попадёт)
        mov DPTR, #ADR_KB       ; регистр столбцов клавиатуры
        clr A
        movx @DPTR, A           ; опустить все столбцы: любое нажатие сразу даст спад на INT0
        setb IT0                ; INT0 срабатывает по спаду
        clr IE0                 ; забыть возможный ложный запрос от включения
        setb EX0                ; разрешить прерывание от клавиатуры
        setb EA                 ; и прерывания вообще
        jmp $ ; %stop%

; KbScan — опрос матрицы 4x3 по столбцам, вызывается из обработчика INT0.
; Каждый столбец опускается по очереди, при этом читаются три строки.
; Считаем, сколько контактов замкнуто: ровно один — код по табл. 3 ТЗ, иначе 0FFh.
; Вход: нет. Выход: A — код клавиши (0-9, Т = 10, С = 11) или 0FFh.
; Портит: только A и флаги; R2-R6, B, DPTR сохраняются в стеке.
KbScan:
        push PSW
        push DPL
        push DPH
        push AR2
        push AR3
        push AR4
        push AR5
        push AR6
        mov DPTR, #ADR_KB       ; одна ячейка: запись — столбцы, чтение — строки
        mov R3, #0              ; номер клавиши (строка * 4 + столбец) последнего замыкания
        mov R4, #0              ; сколько замыканий нашли за весь опрос
        mov R5, #0FEh           ; маска: ноль в бите опрашиваемого столбца
        mov R2, #0              ; номер текущего столбца, 0..3
KbCol:
        mov A, R5
        movx @DPTR, A           ; опустить только этот столбец
        movx A, @DPTR           ; снять состояние строк
        anl A, #(1 SHL KB_ROWS) - 1 ; старшие биты шины не подключены — отбросить
        mov R6, AR2             ; номер клавиши в первой строке равен номеру столбца
        mov B, #KB_ROWS         ; сколько строк перебрать
KbRow:
        rrc A                   ; бит очередной строки уходит в C
        jc KbRowNext            ; единица — контакт разомкнут
        inc R4                  ; нашли ещё одно замыкание
        mov R3, AR6             ; запомнить, какая это клавиша
KbRowNext:
        xch A, R6               ; номер клавиши для строки ниже
        add A, #KB_COLS         ; больше на число столбцов
        xch A, R6
        djnz B, KbRow
        mov A, R5               ; ноль в маске переносим на следующий столбец
        rl A
        mov R5, A
        inc R2
        cjne R2, #KB_COLS, KbCol
        clr A
        movx @DPTR, A           ; опрос окончен — снова опустить все столбцы для INT0
        cjne R4, #1, KbNone     ; ни одной или две и больше — ошибка
        mov A, R3
        mov DPTR, #KbCodes
        movc A, @A+DPTR         ; номер клавиши -> код по табл. 3 ТЗ
        sjmp KbOut
KbNone:
        mov A, #0FFh            ; помеха или несколько клавиш сразу
KbOut:
        clr IE0                 ; пока перебирали столбцы, INT0 дёргался — сбросить лишний запрос
        pop AR6
        pop AR5
        pop AR4
        pop AR3
        pop AR2
        pop DPH
        pop DPL
        pop PSW
        ret

; Коды клавиш по рис. 4 ТЗ (4 столбца x 3 строки), строка за строкой
KbCodes:
        DB 1, 2, 3, 10          ; Row1: 1 2 3 Т
        DB 4, 5, 6, 11          ; Row2: 4 5 6 С
        DB 7, 8, 9, 0           ; Row3: 7 8 9 0
END
