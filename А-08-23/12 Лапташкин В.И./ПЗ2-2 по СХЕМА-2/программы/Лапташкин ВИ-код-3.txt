; Лапташкин В.И., А-08-23, 12, v1
; ---- vars.inc (вклеен mpscode) ----
; Параметры варианта А-08, M=12 (ТЗ 2026). Сгенерировано mpsgen — не править руками.

VAR_M        EQU 12            ; номер варианта
VAR_G        EQU 8            ; номер группы
Y2_CONST     EQU 20            ; G+M: Y2 = (Y2_CONST + X1 + X2) mod 256

; --- адреса внешних устройств (movx) ---
ADR_IND      EQU 0A000h        ; индикатор, CS Y5
ADR_KB       EQU 0C000h        ; клавиатура: запись столбцов / чтение строк, CS Y6
ADR_Y2       EQU 8000h        ; регистр Y2, CS Y4
BUF_START    EQU 6000h        ; первая ячейка буфера, CS Y3
BUF_END      EQU 65F8h        ; первая ячейка ЗА буфером (точка заворота)
BUF_SIZE     EQU 1528

; --- клавиатура 4x3 и индикатор ---
KB_COLS      EQU 4             ; столбцы Col1…Col4 — биты D0…D3 при записи в ADR_KB (0 — столбец опрашивается)
KB_ROWS      EQU 3             ; строки Row1…Row3 — биты D0…D2 при чтении ADR_KB (0 — замкнута); старшие биты не определены — маскировать
IND_OFF      EQU 0FFh           ; индикатор погашен (общий анод: 0 зажигает сегмент; D0…D7 = a…g, dp)

; --- внутреннее ОЗУ ---
HEAD_L       EQU 40h          ; «голова», младший байт
HEAD_H       EQU 41h          ; «голова», старший байт
TAIL_L       EQU 42h          ; «хвост», младший байт
TAIL_H       EQU 43h          ; «хвост», старший байт
F_EMPTY      BIT 0Ch          ; флаг пустоты буфера (21h.4)
F_OVF        BIT 14h          ; флаг переполнения буфера (22h.4)

; --- выводы ---
PIN_Y1       BIT P1.5         ; строб Y1, активный 0
PIN_Y2       BIT P1.6         ; строб Y2, активный 0
PIN_CSEN     BIT P3.4         ; CS_EN: разрешение дешифратора CS (1 — CS активны)

; --- таймеры (кварц 12 МГц) ---
T2_RELOAD_H  EQU 0F7h          ; строб Y2 2068 мкс, Timer1
T2_RELOAD_L  EQU 0ECh
TICK_H       EQU 0F8h          ; тик 2 мс для Y1/T3, Timer0
TICK_L       EQU 30h
Y1_TICKS     EQU 149           ; T1 = 298 мс
T3_TICKS     EQU 1410          ; T3 = 2820 мс
; ---- конец vars.inc ----
; Timer0 в режиме 2 (8 бит с автоперезагрузкой): переполнение каждые 200 мкс без программной перезагрузки,
; 10 переполнений = один тик 2 мс из vars.inc, T3 = T3_TICKS таких тиков
SUB_US  EQU 200                 ; период переполнения Timer0, мкс (1 машинный цикл = 1 мкс)
SUB_N   EQU 10                  ; переполнений в одном тике 2 мс
DSEG AT 30h
SubCnt: DS 1                    ; сколько переполнений по 200 мкс осталось до конца тика
TicksL: DS 1                    ; счётчик тиков T3 для djnz: младший байт
TicksH: DS 1                    ; старший байт (сколько раз пройти младший до нуля)
CSEG AT 0h
        sjmp START
org 03h ; "заглушка" для INT0
        nop
        reti
org 0bh                         ; Timer0 — тик для отсчёта T3
        ljmp T0Isr
org 13h ; "заглушка" для INT1
        nop
        reti
org 1bh ; "заглушка" для Tmr1_ovf
        nop
        reti
org 23h ; "заглушка" для UART
        nop
        reti
org 2bh
START:
; Инициализация МК **********************************************
        mov SP, #07h            ; стек сразу за банком 0
        setb PIN_CSEN           ; разрешить дешифратор CS
        mov DPTR, #ADR_IND
        mov A, #IND_OFF
        movx @DPTR, A           ; при включении индикатор тёмный
        mov TMOD, #02h          ; Timer0 — режим 2, перезагрузка из TH0 аппаратно
        mov TH0, #256 - SUB_US  ; значение перезагрузки: 200 счётов до переполнения
        clr TR0                 ; пока символа нет — таймер стоит
        clr TF0
        setb ET0                ; прерывание от Timer0
        setb EA
        mov A, #5               ; пример: показать «5»
        call Show ; %proc%
        jmp $ ; %stop%

; Show — показать символ на индикаторе и запустить отсчёт T3 до гашения.
; Повторный вызов перезапускает T3 заново. Коды больше 10 гасят индикатор сразу.
; Вход: A — код по табл. 2 ТЗ (0-9 — цифра, 10 — «E», больше — погасить).
; Выход: нет (символ в регистре индикатора, Timer0 считает T3).
; Портит: A и флаги; DPTR сохраняется в стеке.
Show:
        push DPL
        push DPH
        clr TR0                 ; прежний отсчёт T3 больше не нужен
        clr TF0                 ; и его недосчитанный запрос тоже
        cjne A, #11, ShCmp      ; C = 1, если код меньше 11
ShCmp:
        jnc ShOff               ; кода такого нет — просто погасить
        mov DPTR, #SegTab
        movc A, @A+DPTR         ; сегменты для общего катода (1 — горит)
        xrl A, #IND_OFF         ; общий анод: сегмент зажигает 0, поэтому инвертируем
        mov DPTR, #ADR_IND
        movx @DPTR, A           ; символ в регистр индикатора
        mov SubCnt, #SUB_N      ; новый тик начинается с нуля
        mov TicksL, #LOW(T3_TICKS)        ; счёт под djnz: сначала младший байт до нуля,
        mov TicksH, #HIGH(T3_TICKS - 1) + 1 ; потом по 256 на каждую единицу старшего
        mov TL0, #256 - SUB_US  ; первое переполнение через 200 мкс
        setb TR0                ; пошёл отсчёт
        sjmp ShRet
ShOff:
        mov DPTR, #ADR_IND
        mov A, #IND_OFF
        movx @DPTR, A           ; погасить все сегменты
ShRet:
        pop DPH
        pop DPL
        ret

; T0Isr — переполнение Timer0 раз в 200 мкс: каждые 10 раз — тик 2 мс, счёт тиков T3.
; Когда тики кончились — индикатор гасится, таймер останавливается.
; djnz флаги не меняет, поэтому PSW трогать не нужно; A сохраняется там, где используется.
; Вход: SubCnt, TicksH:TicksL. Выход: нет. Портит: ничего.
T0Isr:
        djnz SubCnt, T0Exit     ; тик 2 мс ещё не набрался
        mov SubCnt, #SUB_N      ; следующий тик
        djnz TicksL, T0Exit     ; минус один тик
        djnz TicksH, T0Exit     ; младший байт прошёл круг — минус 256 тиков
        clr TR0                 ; T3 истекло — больше не считать
        push ACC
        push DPL
        push DPH
        mov DPTR, #ADR_IND
        mov A, #IND_OFF
        movx @DPTR, A           ; гасим индикатор
        pop DPH
        pop DPL
        pop ACC
T0Exit:
        reti

; Сегменты цифр 0-9 и «E» для общего катода: D0..D7 = a, b, c, d, e, f, g, dp
SegTab:
        DB 3Fh, 06h, 5Bh, 4Fh, 66h  ; 0 1 2 3 4
        DB 6Dh, 7Dh, 07h, 7Fh, 6Fh  ; 5 6 7 8 9
        DB 79h                      ; E
END
