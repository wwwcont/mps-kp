; Хромов И., А-08-23, 20, v1
; ---- vars.inc (вклеен mpscode) ----
; Параметры варианта А-08, M=20 (ТЗ 2026). Сгенерировано mpsgen — не править руками.

VAR_M        EQU 20            ; номер варианта
VAR_G        EQU 8            ; номер группы
Y2_CONST     EQU 28            ; G+M: Y2 = (Y2_CONST + X1 + X2) mod 256

; --- адреса внешних устройств (movx) ---
ADR_IND      EQU 0A000h        ; индикатор, CS Y5
ADR_KB       EQU 0C000h        ; клавиатура: запись столбцов / чтение строк, CS Y6
ADR_Y2       EQU 8000h        ; регистр Y2, CS Y4
BUF_START    EQU 6000h        ; первая ячейка буфера, CS Y3
BUF_END      EQU 6648h        ; первая ячейка ЗА буфером (точка заворота)
BUF_SIZE     EQU 1608

; --- клавиатура 4x3 и индикатор ---
KB_COLS      EQU 4             ; столбцы Col1…Col4 — биты D0…D3 при записи в ADR_KB (0 — столбец опрашивается)
KB_ROWS      EQU 3             ; строки Row1…Row3 — биты D0…D2 при чтении ADR_KB (0 — замкнута); старшие биты не определены — маскировать
IND_OFF      EQU 0FFh           ; индикатор погашен (общий анод: 0 зажигает сегмент; D0…D7 = a…g, dp)

; --- внутреннее ОЗУ ---
HEAD_L       EQU 40h          ; «голова», младший байт
HEAD_H       EQU 41h          ; «голова», старший байт
TAIL_L       EQU 42h          ; «хвост», младший байт
TAIL_H       EQU 43h          ; «хвост», старший байт
F_EMPTY      BIT 14h          ; флаг пустоты буфера (22h.4)
F_OVF        BIT 1Ch          ; флаг переполнения буфера (23h.4)

; --- выводы ---
PIN_Y1       BIT P1.6         ; строб Y1, активный 0
PIN_Y2       BIT P1.7         ; строб Y2, активный 0
PIN_CSEN     BIT P3.4         ; CS_EN: разрешение дешифратора CS (1 — CS активны)

; --- таймеры (кварц 12 МГц) ---
T2_RELOAD_H  EQU 0F6h          ; строб Y2 2308 мкс, Timer1
T2_RELOAD_L  EQU 0FCh
TICK_H       EQU 0C9h          ; тик 14 мс для Y1/T3, Timer0
TICK_L       EQU 50h
Y1_TICKS     EQU 27           ; T1 = 378 мс
T3_TICKS     EQU 230          ; T3 = 3220 мс
; ---- конец vars.inc ----
DSEG AT 30h
X1:     DS 1                    ; цифра с клавиатуры (0 — Y2 пассивен)
X2:     DS 1                    ; отсчёт из буфера
CSEG AT 0h
        sjmp START
org 03h ; "заглушка" для INT0
        nop
        reti
org 0bh ; "заглушка" для Tmr0_ovf
        nop
        reti
org 13h ; "заглушка" для INT1
        nop
        reti
org 1bh                         ; Timer1 — конец строба Y2
        clr TR1 ; остановить таймер после переполнения
        setb PIN_Y2 ; вернуть строб на пассивный уровень
        reti
org 23h ; "заглушка" для UART
        nop
        reti
org 2bh
START:
; Инициализация МК **********************************************
        mov SP, #07h
        setb PIN_CSEN ; разрешить запись во внешний регистр
        setb PIN_Y2 ; вернуть строб на пассивный уровень
        mov TMOD, #10h ; таймер 1 работает в 16-разрядном режиме
        setb PT1 ; прерывание конца строба имеет высокий приоритет
        setb ET1
        setb EA
        mov X1, #3              ; пример входных данных
        mov X2, #200
        call Y2Out ; %proc%
        jmp $ ; %stop%

; Y2Out — расчёт Y2 = (G+M+X1+X2) mod 256 (X1 = 0 → Y2 = 0), запись в регистр Y2 и строб T2.
; Вход: X1, X2. Выход: регистр Y2; PIN_Y2 = 0 на T2 мкс (конец — в прерывании Timer1).
; Изменяет A, DPTR и флаги арифметики.
Y2Out:
        mov A, X1 ; нулевой X1 сразу даёт нулевой выход
        jz WriteY2
        add A, X2
        add A, #Y2_CONST ; сложение в аккумуляторе даёт остаток по модулю 256
WriteY2:
        mov DPTR, #ADR_Y2
        movx @DPTR, A ; установить код до активного фронта строба
        clr TR1 ; остановить таймер после переполнения
        clr TF1 ; снять старый запрос переполнения
        mov TH1, #T2_RELOAD_H ; загрузить старший байт начального счёта
        mov TL1, #T2_RELOAD_L ; загрузить младший байт начального счёта
        clr PIN_Y2 ; выдать активный низкий уровень
        setb TR1 ; начать отсчёт длительности строба
        ret
END
