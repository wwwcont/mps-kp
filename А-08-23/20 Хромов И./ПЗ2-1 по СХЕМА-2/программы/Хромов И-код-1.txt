; Хромов И., А-08-23, 20, v1
; ---- vars.inc (вклеен mpscode) ----
; Параметры варианта А-08, M=20 (ТЗ 2026). Сгенерировано mpsgen — не править руками.

VAR_M        EQU 20            ; номер варианта
VAR_G        EQU 8            ; номер группы
Y2_CONST     EQU 28            ; G+M: Y2 = (Y2_CONST + X1 + X2) mod 256

; --- адреса внешних устройств (movx) ---
ADR_IND      EQU 0A000h        ; индикатор, CS Y5
ADR_KB       EQU 0C000h        ; клавиатура: запись столбцов / чтение строк, CS Y6
ADR_Y2       EQU 8000h        ; регистр Y2, CS Y4
BUF_START    EQU 6000h        ; первая ячейка буфера, CS Y3
BUF_END      EQU 6648h        ; первая ячейка ЗА буфером (точка заворота)
BUF_SIZE     EQU 1608

; --- клавиатура 4x3 и индикатор ---
KB_COLS      EQU 4             ; столбцы Col1…Col4 — биты D0…D3 при записи в ADR_KB (0 — столбец опрашивается)
KB_ROWS      EQU 3             ; строки Row1…Row3 — биты D0…D2 при чтении ADR_KB (0 — замкнута); старшие биты не определены — маскировать
IND_OFF      EQU 0FFh           ; индикатор погашен (общий анод: 0 зажигает сегмент; D0…D7 = a…g, dp)

; --- внутреннее ОЗУ ---
HEAD_L       EQU 40h          ; «голова», младший байт
HEAD_H       EQU 41h          ; «голова», старший байт
TAIL_L       EQU 42h          ; «хвост», младший байт
TAIL_H       EQU 43h          ; «хвост», старший байт
F_EMPTY      BIT 14h          ; флаг пустоты буфера (22h.4)
F_OVF        BIT 1Ch          ; флаг переполнения буфера (23h.4)

; --- выводы ---
PIN_Y1       BIT P1.6         ; строб Y1, активный 0
PIN_Y2       BIT P1.7         ; строб Y2, активный 0
PIN_CSEN     BIT P3.4         ; CS_EN: разрешение дешифратора CS (1 — CS активны)

; --- таймеры (кварц 12 МГц) ---
T2_RELOAD_H  EQU 0F6h          ; строб Y2 2308 мкс, Timer1
T2_RELOAD_L  EQU 0FCh
TICK_H       EQU 0C9h          ; тик 14 мс для Y1/T3, Timer0
TICK_L       EQU 50h
Y1_TICKS     EQU 27           ; T1 = 378 мс
T3_TICKS     EQU 230          ; T3 = 3220 мс
; ---- конец vars.inc ----
DSEG AT 30h

CSEG AT 0h
        sjmp START
org 03h                         ; INT0 — нажатие клавиши: процедура вызывается отсюда
        call KbScan ; %proc%
        reti
org 0bh ; "заглушка" для Tmr0_ovf
        nop
        reti
org 13h ; "заглушка" для INT1
        nop
        reti
org 1bh ; "заглушка" для Tmr1_ovf
        nop
        reti
org 23h ; "заглушка" для UART
        nop
        reti
org 2bh
START:
; Инициализация МК **********************************************
        mov SP, #07h
        setb PIN_CSEN ; разрешить дешифратор внешних устройств
        mov DPTR, #ADR_KB ; адрес регистра столбцов и буфера строк
        clr A ; нулевой код включает все столбцы
        movx @DPTR, A ; выставить маску столбцов
        setb IT0 ; внешнее прерывание по спаду
        clr IE0 ; снять запрос, возникший при переключении столбцов
        setb EX0
        setb EA
        jmp $ ; %stop%

; KbScan — сканирование матрицы клавиатуры по столбцам (обработчик INT0).
; Вход: нет. Выход: A — код единственной нажатой клавиши по табл. 3 ТЗ или 0FFh.
; Изменяет A, сохраняет PSW, DPTR и R2–R7 (банк 0).
KbScan:
        push PSW ; сохранить состояние прерванного кода
        push DPL ; сохранить младший байт указателя
        push DPH ; сохранить старший байт указателя
        push 02h ; сохранить R2–R7 из банка 0
        push 03h
        push 04h
        push 05h
        push 06h
        push 07h

        mov R2, #0FEh           ; ноль только в опрашиваемом столбце
        mov R3, #KB_COLS ; опросить все столбцы матрицы
        mov R4, #0              ; индекс контакта в таблице
        mov R5, #0              ; число замкнутых контактов
        mov R7, #0FFh           ; результат, если замыканий нет
ScanCol:
        mov DPTR, #ADR_KB ; адрес регистра столбцов и буфера строк
        mov A, R2
        movx @DPTR, A ; выставить маску столбцов
        movx A, @DPTR ; считать состояния строк
        mov R6, #KB_ROWS ; проверять только используемые разряды
ScanRow:
        rrc A ; перенести очередную строку в CY
        jc NextRow ; единица означает разомкнутую строку
        inc R5
        cjne R5, #1, Several ; второе замыкание уже исключает единственную клавишу
        push ACC ; сохранить непрочитанные разряды строк
        mov DPTR, #KeyCodes
        mov A, R4
        movc A, @A+DPTR ; найти код по номеру контакта
        mov R7, A
        pop ACC ; продолжить опрос той же колонки
NextRow:
        inc R4
        djnz R6, ScanRow
        mov A, R2
        rl A ; перенести ноль на следующий столбец
        mov R2, A
        djnz R3, ScanCol
        sjmp ScanDone
Several:
        mov R7, #0FFh
ScanDone:
        mov DPTR, #ADR_KB ; адрес регистра столбцов и буфера строк
        clr A ; нулевой код включает все столбцы
        movx @DPTR, A           ; вернуть все столбцы в ноль
        clr IE0 ; снять запрос, возникший при переключении столбцов
        mov A, R7 ; вернуть код через аккумулятор
        pop 07h ; восстановить R7–R2
        pop 06h
        pop 05h
        pop 04h
        pop 03h
        pop 02h
        pop DPH
        pop DPL
        pop PSW ; восстановить флаги прерванной программы
        ret

KeyCodes:
        DB 1, 4, 7, 2, 5, 8, 3, 6, 9, 10, 11, 0
END
