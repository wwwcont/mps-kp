; Грошев А.В., А-06-23, 1, v1
; ---- vars.inc (вклеен mpscode) ----
; Параметры варианта А-06, M=1 (ТЗ 2026). Сгенерировано mpsgen — не править руками.

VAR_M        EQU 1            ; номер варианта
VAR_G        EQU 6            ; номер группы
Y2_CONST     EQU 7            ; G+M: Y2 = (Y2_CONST + X1 + X2) mod 256

; --- адреса внешних устройств (movx) ---
ADR_IND      EQU 4000h        ; индикатор, CS Y2
ADR_KB       EQU 8000h        ; клавиатура: запись столбцов / чтение строк, CS Y4
ADR_Y2       EQU 6000h        ; регистр Y2, CS Y3
BUF_START    EQU 2000h        ; первая ячейка буфера, CS Y1
BUF_END      EQU 24BAh        ; первая ячейка ЗА буфером (точка заворота)
BUF_SIZE     EQU 1210
BUF_X2IN     EQU 3FFFh        ; ячейка обмена: сюда внешнее устройство пишет X2 правым портом (строб X2 = прерывание INT1)

; --- клавиатура 3x4 и индикатор ---
KB_COLS      EQU 3             ; столбцы Col1…Col3 — биты D0…D2 при записи в ADR_KB (0 — столбец опрашивается)
KB_ROWS      EQU 4             ; строки Row1…Row4 — биты D0…D3 при чтении ADR_KB (0 — замкнута); старшие биты не определены — маскировать
IND_OFF      EQU 00h           ; индикатор погашен (общий катод: 1 зажигает сегмент; D0…D7 = a…g, dp)

; --- внутреннее ОЗУ ---
HEAD_L       EQU 38h          ; «голова», младший байт
HEAD_H       EQU 39h          ; «голова», старший байт
TAIL_L       EQU 3Ah          ; «хвост», младший байт
TAIL_H       EQU 3Bh          ; «хвост», старший байт
F_EMPTY      BIT 1h          ; флаг пустоты буфера (20h.1)
F_OVF        BIT 7h          ; флаг переполнения буфера (20h.7)

; --- выводы ---
PIN_Y1       BIT P1.1         ; строб Y1, активный 0
PIN_Y2       BIT P1.2         ; строб Y2, активный 0
PIN_CSEN     BIT P3.4         ; CS_EN: разрешение дешифратора CS (1 — CS активны)

; --- таймеры (кварц 12 МГц) ---
T2_RELOAD_H  EQU 0FAh          ; строб Y2 1536 мкс, Timer1
T2_RELOAD_L  EQU 0h
TICK_H       EQU 0F8h          ; тик 2 мс для Y1/T3, Timer0
TICK_L       EQU 30h
Y1_TICKS     EQU 88           ; T1 = 176 мс
T3_TICKS     EQU 1075          ; T3 = 2150 мс
; ---- конец vars.inc ----
DSEG AT 30h
; переменные в ОЗУ не нужны: всё держится в регистрах банка 0
CSEG AT 0h
        sjmp START
org 03h                         ; INT0 — нажатие клавиши: процедура вызывается отсюда
        call KbScan ; %proc%
        reti
org 0bh ; "заглушка" для Tmr0_ovf
        nop
        reti
org 13h ; "заглушка" для INT1
        nop
        reti
org 1bh ; "заглушка" для Tmr1_ovf
        nop
        reti
org 23h ; "заглушка" для UART
        nop
        reti
org 2bh
START:
; Инициализация МК **********************************************
        mov SP, #07h
        setb PIN_CSEN           ; разрешаем дешифратор адреса, иначе клавиатура не выбирается
        mov DPTR, #ADR_KB
        clr A
        movx @DPTR, A           ; опускаем все столбцы: любое нажатие сразу потянет INT0 вниз
        setb IT0                ; INT0 срабатывает по спаду
        clr IE0                 ; забываем возможный ложный запрос до разрешения
        setb EX0                ; разрешаем прерывание от клавиатуры
        setb EA                 ; и прерывания вообще
        jmp $ ; %stop%

; KbScan — опрос матрицы 3x4 по столбцам (вызывается из обработчика INT0).
; Назначение: найти единственную нажатую клавишу и вернуть её код по табл. 3 ТЗ.
; Вход: нет.
; Выход: A — код клавиши (0…9, «Т» = 10, «С» = 11) или 0FFh, если нажато
;        больше одной клавиши или ни одной (помеха).
; Портит: только A; B, DPTR, PSW, R2–R7 сохраняются в стеке.
KbScan:
        push B
        push PSW
        push DPL
        push DPH
        push AR2
        push AR3
        push AR4
        push AR5
        push AR6
        push AR7
        mov DPTR, #ADR_KB
        mov R2, #0              ; R2 — сколько замкнутых контактов нашли
        mov R3, #0              ; R3 — номер текущего столбца с нуля
        mov R4, #0FEh           ; R4 — маска столбцов: ноль только у опрашиваемого
KsCol:
        mov A, R4
        movx @DPTR, A           ; опускаем один столбец, остальные держим в 1
        movx A, @DPTR           ; снимаем состояние строк
        anl A, #(1 SHL KB_ROWS) - 1   ; оставляем только подключённые строки
        xrl A, #(1 SHL KB_ROWS) - 1   ; теперь 1 — строка замкнута на этот столбец
        mov R5, A               ; R5 — ещё не разобранные замкнутые строки
        mov A, R3
        mov R6, A               ; R6 — индекс клавиши в таблице: строка*KB_COLS + столбец
KsRow:
        mov A, R5
        jz KsNext               ; замкнутых строк в этом столбце больше нет
        clr C
        rrc A                   ; младшая строка уходит в перенос
        mov R5, A
        jnc KsSkip              ; эта строка не замкнута
        inc R2                  ; ещё одно нажатие
        mov A, R6
        mov R7, A               ; запоминаем индекс найденной клавиши
KsSkip:
        mov A, R6
        add A, #KB_COLS         ; переходим к той же колонке строкой ниже
        mov R6, A
        sjmp KsRow
KsNext:
        mov A, R4
        rl A                    ; ноль маски сдвигается на следующий столбец
        mov R4, A
        inc R3
        cjne R3, #KB_COLS, KsCol
        clr A
        movx @DPTR, A           ; снова все столбцы в 0 — ждём следующего нажатия
        clr IE0                 ; во время опроса линия INT0 дёргалась — лишний запрос не нужен
        cjne R2, #1, KsBad      ; принимаем только одну нажатую клавишу
        mov A, R7
        mov DPTR, #KeyTab
        movc A, @A+DPTR         ; индекс в матрице -> код клавиши
        sjmp KsOut
KsBad:
        mov A, #0FFh            ; ничего не нажато или нажато несколько
KsOut:
        pop AR7
        pop AR6
        pop AR5
        pop AR4
        pop AR3
        pop AR2
        pop DPH
        pop DPL
        pop PSW
        pop B
        ret

; Коды клавиш по рис. 4 ТЗ (3x4): строки сверху вниз, в строке столбцы слева направо
KeyTab:
        DB 1, 2, 3
        DB 4, 5, 6
        DB 7, 8, 9
        DB 0, 10, 11
END
