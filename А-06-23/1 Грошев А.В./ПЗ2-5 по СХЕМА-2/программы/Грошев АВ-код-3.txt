; Грошев А.В., А-06-23, 1, v1
; ---- vars.inc (вклеен mpscode) ----
; Параметры варианта А-06, M=1 (ТЗ 2026). Сгенерировано mpsgen — не править руками.

VAR_M        EQU 1            ; номер варианта
VAR_G        EQU 6            ; номер группы
Y2_CONST     EQU 7            ; G+M: Y2 = (Y2_CONST + X1 + X2) mod 256

; --- адреса внешних устройств (movx) ---
ADR_IND      EQU 4000h        ; индикатор, CS Y2
ADR_KB       EQU 8000h        ; клавиатура: запись столбцов / чтение строк, CS Y4
ADR_Y2       EQU 6000h        ; регистр Y2, CS Y3
BUF_START    EQU 2000h        ; первая ячейка буфера, CS Y1
BUF_END      EQU 24BAh        ; первая ячейка ЗА буфером (точка заворота)
BUF_SIZE     EQU 1210
BUF_X2IN     EQU 3FFFh        ; ячейка обмена: сюда внешнее устройство пишет X2 правым портом (строб X2 = прерывание INT1)

; --- клавиатура 3x4 и индикатор ---
KB_COLS      EQU 3             ; столбцы Col1…Col3 — биты D0…D2 при записи в ADR_KB (0 — столбец опрашивается)
KB_ROWS      EQU 4             ; строки Row1…Row4 — биты D0…D3 при чтении ADR_KB (0 — замкнута); старшие биты не определены — маскировать
IND_OFF      EQU 00h           ; индикатор погашен (общий катод: 1 зажигает сегмент; D0…D7 = a…g, dp)

; --- внутреннее ОЗУ ---
HEAD_L       EQU 38h          ; «голова», младший байт
HEAD_H       EQU 39h          ; «голова», старший байт
TAIL_L       EQU 3Ah          ; «хвост», младший байт
TAIL_H       EQU 3Bh          ; «хвост», старший байт
F_EMPTY      BIT 1h          ; флаг пустоты буфера (20h.1)
F_OVF        BIT 7h          ; флаг переполнения буфера (20h.7)

; --- выводы ---
PIN_Y1       BIT P1.1         ; строб Y1, активный 0
PIN_Y2       BIT P1.2         ; строб Y2, активный 0
PIN_CSEN     BIT P3.4         ; CS_EN: разрешение дешифратора CS (1 — CS активны)

; --- таймеры (кварц 12 МГц) ---
T2_RELOAD_H  EQU 0FAh          ; строб Y2 1536 мкс, Timer1
T2_RELOAD_L  EQU 0h
TICK_H       EQU 0F8h          ; тик 2 мс для Y1/T3, Timer0
TICK_L       EQU 30h
Y1_TICKS     EQU 88           ; T1 = 176 мс
T3_TICKS     EQU 1075          ; T3 = 2150 мс
; ---- конец vars.inc ----
DSEG AT 30h
Y1Left: DS 1                    ; сколько тиков по 2 мс осталось до конца строба Y1
CSEG AT 0h
        sjmp START
org 03h ; "заглушка" для INT0
        nop
        reti
org 0bh                         ; Timer0 — тик для отсчёта T1
        ljmp T0Isr
org 13h ; "заглушка" для INT1
        nop
        reti
org 1bh ; "заглушка" для Tmr1_ovf
        nop
        reti
org 23h ; "заглушка" для UART
        nop
        reti
org 2bh
START:
; Инициализация МК **********************************************
        mov SP, #07h
        setb PIN_CSEN           ; дешифратор адреса разрешён
        setb PIN_Y1             ; строб Y1 в пассивном состоянии
        mov Y1Left, #0
        anl TMOD, #0F0h         ; Timer0 — 16-битный счётчик (режим 1), Timer1 не трогаем
        orl TMOD, #01h
        clr TR0                 ; таймер пока стоит
        clr TF0
        setb ET0                ; прерывание по переполнению Timer0
        setb EA
        call Y1Start ; %proc%
        jmp $ ; %stop%

; Y1Start — запуск строба Y1 длительностью T1.
; Назначение: опустить PIN_Y1 в 0 и запустить Timer0 на Y1_TICKS тиков по 2 мс;
;   строб поднимает обработчик Timer0, процедура сразу возвращается.
; Вход: нет. Выход: PIN_Y1 = 0, Timer0 запущен, Y1Left = Y1_TICKS.
; Портит: ничего (меняет только TH0, TL0, TF0, TR0).
Y1Start:
        clr TR0                 ; повторный вызов начинает отсчёт заново
        mov TH0, #TICK_H        ; первый тик — ровно 2 мс
        mov TL0, #TICK_L
        clr TF0
        mov Y1Left, #Y1_TICKS
        clr PIN_Y1              ; начало строба
        setb TR0
        ret

; T0Isr — обработчик Timer0: прошёл очередной тик 2 мс.
; Назначение: перезарядить Timer0 с учётом тактов, натикавших после переполнения,
;   уменьшить счётчик и по его обнулению закончить строб Y1.
; Вход: Y1Left. Выход: Y1Left - 1; при нуле PIN_Y1 = 1 и Timer0 остановлен.
; Портит: ничего (A и PSW в стеке).
T0Isr:
        push ACC
        push PSW
        clr TR0                 ; на время перезарядки таймер стоит (7 тактов)
        mov A, TL0
        add A, #LOW((TICK_H SHL 8) + TICK_L + 7)   ; добавляем к уже натикавшему и
        mov TL0, A                                 ; учитываем 7 тактов остановки
        mov A, TH0
        addc A, #HIGH((TICK_H SHL 8) + TICK_L + 7)
        mov TH0, A
        setb TR0
        djnz Y1Left, T0Out      ; строб ещё идёт
        clr TR0                 ; T1 истекло: таймер больше не нужен
        setb PIN_Y1             ; конец строба
T0Out:
        pop PSW
        pop ACC
        reti
END
