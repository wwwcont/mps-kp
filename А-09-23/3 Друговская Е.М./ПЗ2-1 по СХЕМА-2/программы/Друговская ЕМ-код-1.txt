; Параметры варианта А-09, M=3 (ТЗ 2026). Сгенерировано mpsgen — не править руками.

VAR_M        EQU 3            ; номер варианта
VAR_G        EQU 9            ; номер группы
Y2_CONST     EQU 12            ; G+M: Y2 = (Y2_CONST + X1 + X2) mod 256

; --- адреса внешних устройств (movx) ---
ADR_IND      EQU 0A000h        ; индикатор, CS Y5
ADR_KB       EQU 0E000h        ; клавиатура: запись столбцов / чтение строк, CS Y7
ADR_Y2       EQU 0C000h        ; регистр Y2, CS Y6
BUF_START    EQU 8000h        ; первая ячейка буфера, CS Y4
BUF_END      EQU 85FAh        ; первая ячейка ЗА буфером (точка заворота)
BUF_SIZE     EQU 1530
BUF_X2IN     EQU 9FFFh        ; ячейка обмена: сюда внешнее устройство пишет X2 правым портом (строб X2 = прерывание INT1)

; --- клавиатура 3x4 и индикатор ---
KB_COLS      EQU 3             ; столбцы Col1…Col3 — биты D0…D2 при записи в ADR_KB (0 — столбец опрашивается)
KB_ROWS      EQU 4             ; строки Row1…Row4 — биты D0…D3 при чтении ADR_KB (0 — замкнута); старшие биты не определены — маскировать
IND_OFF      EQU 0FFh           ; индикатор погашен (общий анод: 0 зажигает сегмент; D0…D7 = a…g, dp)

; --- внутреннее ОЗУ ---
HEAD_L       EQU 44h          ; «голова», младший байт
HEAD_H       EQU 45h          ; «голова», старший байт
TAIL_L       EQU 46h          ; «хвост», младший байт
TAIL_H       EQU 47h          ; «хвост», старший байт
F_EMPTY      BIT 3h          ; флаг пустоты буфера (20h.3)
F_OVF        BIT 0Ch          ; флаг переполнения буфера (21h.4)

; --- выводы ---
PIN_Y1       BIT P1.3         ; строб Y1, активный 0
PIN_Y2       BIT P1.4         ; строб Y2, активный 0
PIN_CSEN     BIT P3.4         ; CS_EN: разрешение дешифратора CS (1 — CS активны)

; --- таймеры (кварц 12 МГц) ---
T2_RELOAD_H  EQU 0F8h          ; строб Y2 1899 мкс, Timer1
T2_RELOAD_L  EQU 95h
TICK_H       EQU 0ECh          ; тик 5 мс для Y1/T3, Timer0
TICK_L       EQU 78h
Y1_TICKS     EQU 43           ; T1 = 215 мс
T3_TICKS     EQU 486          ; T3 = 2430 мс
; ---- конец vars.inc ----
DSEG AT 30h
; здесь объявление переменных (если нужны)
CSEG AT 0h
        sjmp START
org 03h                         ; INT0 — нажатие клавиши: процедура вызывается отсюда
        call KbScan ; %proc%
        reti
org 0bh ; "заглушка" для Tmr0_ovf
        nop
        reti
org 13h ; "заглушка" для INT1
        nop
        reti
org 1bh ; "заглушка" для Tmr1_ovf
        nop
        reti
org 23h ; "заглушка" для UART
        nop
        reti
org 2bh
START:
; Инициализация МК **********************************************
        mov SP, #07h
        ; TODO: CS_EN = 1; все столбцы в 0 (иначе нажатие не опустит строку и INT0);
        ; TODO: режим INT0 (IT0), разрешить EX0 и EA
        jmp $ ; %stop%

; KbScan — сканирование матрицы клавиатуры по столбцам (обработчик INT0).
; Вход: нет. Выход: A — код единственной нажатой клавиши по табл. 3 ТЗ или 0FFh.
; TODO: портит …
KbScan:
        ; TODO: по очереди 0 в один столбец (запись в ADR_KB), чтение строк (младшие KB_ROWS бит),
        ; TODO: код по рис. 4; две клавиши и больше — 0FFh; в конце все столбцы снова в 0
        mov A, #0FFh
        ret
END
