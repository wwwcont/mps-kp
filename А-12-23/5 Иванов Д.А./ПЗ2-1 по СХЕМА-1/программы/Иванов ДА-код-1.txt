; Иванов Д.А., А-12-23, 5, v1
; ---- vars.inc (вклеен mpscode) ----
; Параметры варианта А-12, M=5 (ТЗ 2026). Сгенерировано mpsgen — не править руками.

VAR_M        EQU 5            ; номер варианта
VAR_G        EQU 12            ; номер группы
Y2_CONST     EQU 17            ; G+M: Y2 = (Y2_CONST + X1 + X2) mod 256

; --- адреса внешних устройств (movx) ---
ADR_IND      EQU 0C000h        ; индикатор, CS Y6
ADR_KB       EQU 2000h        ; клавиатура: запись столбцов / чтение строк, CS Y1
ADR_Y2       EQU 0E000h        ; регистр Y2, CS Y7
BUF_START    EQU 0A000h        ; первая ячейка буфера, CS Y5
BUF_END      EQU 0A676h        ; первая ячейка ЗА буфером (точка заворота)
BUF_SIZE     EQU 1654
BUF_X2IN     EQU 0BFFFh        ; ячейка обмена: сюда внешнее устройство пишет X2 правым портом (строб X2 = прерывание INT1)

; --- клавиатура 3x4 и индикатор ---
KB_COLS      EQU 3             ; столбцы Col1…Col3 — биты D0…D2 при записи в ADR_KB (0 — столбец опрашивается)
KB_ROWS      EQU 4             ; строки Row1…Row4 — биты D0…D3 при чтении ADR_KB (0 — замкнута); старшие биты не определены — маскировать
IND_OFF      EQU 0FFh           ; индикатор погашен (общий анод: 0 зажигает сегмент; D0…D7 = a…g, dp)

; --- внутреннее ОЗУ ---
HEAD_L       EQU 47h          ; «голова», младший байт
HEAD_H       EQU 48h          ; «голова», старший байт
TAIL_L       EQU 49h          ; «хвост», младший байт
TAIL_H       EQU 4Ah          ; «хвост», старший байт
F_EMPTY      BIT 5h          ; флаг пустоты буфера (20h.5)
F_OVF        BIT 11h          ; флаг переполнения буфера (22h.1)

; --- выводы ---
PIN_Y1       BIT P1.5         ; строб Y1, активный 0
PIN_Y2       BIT P1.6         ; строб Y2, активный 0
PIN_CSEN     BIT P3.4         ; CS_EN: разрешение дешифратора CS (1 — CS активны)

; --- таймеры (кварц 12 МГц) ---
T2_RELOAD_H  EQU 0F7h          ; строб Y2 2062 мкс, Timer1
T2_RELOAD_L  EQU 0F2h
TICK_H       EQU 0FCh          ; тик 1 мс для Y1/T3, Timer0
TICK_L       EQU 18h
Y1_TICKS     EQU 241           ; T1 = 241 мс
T3_TICKS     EQU 2590          ; T3 = 2590 мс
; ---- конец vars.inc ----
DSEG AT 30h
; дополнительные ячейки ОЗУ не нужны: временные значения хранятся в регистрах
CSEG AT 0h
        sjmp START
org 03h                         ; INT0 — нажатие клавиши
        call KbScan ; %proc%
        reti
org 0bh                         ; заглушка Timer0
        nop
        reti
org 13h                         ; заглушка INT1
        nop
        reti
org 1bh                         ; заглушка Timer1
        nop
        reti
org 23h                         ; заглушка UART
        nop
        reti
org 2bh
START:
; Инициализация МК **********************************************
        mov SP, #07h            ; стек начинается после банка регистров 0
        setb PIN_CSEN           ; разрешить дешифратор внешних устройств
        mov DPTR, #ADR_KB       ; адрес регистра столбцов клавиатуры
        clr A                   ; все столбцы держать в нуле для появления INT0
        movx @DPTR, A
        setb IT0                ; INT0 срабатывает по спаду
        clr IE0                 ; убрать возможный запрос от инициализации
        setb EX0                ; разрешить прерывание клавиатуры
        setb EA                 ; общее разрешение прерываний
        jmp $ ; %stop%

; KbScan — опрос всех контактов матрицы 3×4 по столбцам.
; Вход: нет. Выход: A — код единственной клавиши по таблице 3 ТЗ, иначе 0FFh.
; Портит: B, DPTR, R0–R5, PSW. Перед возвратом снова опускает все столбцы.
KbScan:
        mov DPTR, #ADR_KB       ; одна ячейка служит для записи столбцов и чтения строк
        mov R0, #1              ; маска выбранного столбца
        mov R1, #0              ; число найденных замкнутых контактов
        mov R2, #0FFh           ; код первой найденной клавиши
        mov R3, #0              ; номер столбца, начиная с нуля
KB_COLUMN:
        mov A, R0               ; выбранный столбец должен получить нулевой уровень
        cpl A
        anl A, #(1 SHL KB_COLS) - 1
        movx @DPTR, A           ; остальные столбцы временно подняты в единицу
        movx A, @DPTR           ; прочитать активные нулём строки
        anl A, #(1 SHL KB_ROWS) - 1
        cpl A                   ; после инверсии единица означает нажатую клавишу
        anl A, #(1 SHL KB_ROWS) - 1
        mov R5, A               ; сохранить биты строк текущего столбца
        mov R4, #0              ; номер проверяемой строки
KB_ROW:
        mov A, R5
        anl A, #1               ; проверить младший ещё не разобранный бит
        jz KB_NEXT_ROW
        inc R1                  ; учитывается каждое замыкание во всей матрице
        mov A, R1
        cjne A, #1, KB_NEXT_ROW ; код нужен только для первого замыкания
        mov A, R4
        cjne A, #3, KB_DIGIT    ; последняя строка содержит 0, «Т» и «С»
        mov A, R3
        jz KB_SAVE              ; Row4/Col1 соответствует цифре 0
        add A, #9               ; Row4/Col2 и Row4/Col3 дают коды 10 и 11
        sjmp KB_SAVE
KB_DIGIT:
        mov B, #KB_COLS         ; для первых трёх строк код = row*3+col+1
        mul AB
        add A, R3
        inc A
KB_SAVE:
        mov R2, A               ; запомнить код единственного кандидата
KB_NEXT_ROW:
        mov A, R5               ; перейти к следующему биту строки
        clr C
        rrc A
        mov R5, A
        inc R4
        cjne R4, #KB_ROWS, KB_ROW
        mov A, R0               ; выбрать следующий столбец
        rl A
        mov R0, A
        inc R3
        cjne R3, #KB_COLS, KB_COLUMN
        clr A                   ; вернуть все столбцы в ноль до следующего нажатия
        movx @DPTR, A
        clr IE0                 ; сканирование могло создать лишний фронт запроса
        mov A, R1
        cjne A, #1, KB_BAD      ; ноль или несколько клавиш считаются ошибкой
        mov A, R2
        ret
KB_BAD:
        mov A, #0FFh
        ret
END
