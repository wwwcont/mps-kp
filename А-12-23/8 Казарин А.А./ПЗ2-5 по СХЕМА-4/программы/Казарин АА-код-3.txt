; Казарин А.А., А-12-23, 8, v1
; ---- vars.inc (вклеен mpscode) ----
; Параметры варианта А-12, M=8 (ТЗ 2026). Сгенерировано mpsgen — не править руками.

VAR_M        EQU 8            ; номер варианта
VAR_G        EQU 12            ; номер группы
Y2_CONST     EQU 20            ; G+M: Y2 = (Y2_CONST + X1 + X2) mod 256

; --- адреса внешних устройств (movx) ---
ADR_IND      EQU 0E000h        ; индикатор, CS Y7
ADR_KB       EQU 2000h        ; клавиатура: запись столбцов / чтение строк, CS Y1
ADR_Y2       EQU 0C000h        ; регистр Y2, CS Y6
BUF_START    EQU 0A000h        ; первая ячейка буфера, CS Y5
BUF_END      EQU 0A694h        ; первая ячейка ЗА буфером (точка заворота)
BUF_SIZE     EQU 1684
BUF_X2IN     EQU 0BFFFh        ; ячейка обмена: сюда внешнее устройство пишет X2 правым портом (строб X2 = прерывание INT1)

; --- клавиатура 4x3 и индикатор ---
KB_COLS      EQU 4             ; столбцы Col1…Col4 — биты D0…D3 при записи в ADR_KB (0 — столбец опрашивается)
KB_ROWS      EQU 3             ; строки Row1…Row3 — биты D0…D2 при чтении ADR_KB (0 — замкнута); старшие биты не определены — маскировать
IND_OFF      EQU 0FFh           ; индикатор погашен (общий анод: 0 зажигает сегмент; D0…D7 = a…g, dp)

; --- внутреннее ОЗУ ---
HEAD_L       EQU 47h          ; «голова», младший байт
HEAD_H       EQU 48h          ; «голова», старший байт
TAIL_L       EQU 49h          ; «хвост», младший байт
TAIL_H       EQU 4Ah          ; «хвост», старший байт
F_EMPTY      BIT 8h          ; флаг пустоты буфера (21h.0)
F_OVF        BIT 14h          ; флаг переполнения буфера (22h.4)

; --- выводы ---
PIN_Y1       BIT P1.1         ; строб Y1, активный 0
PIN_Y2       BIT P1.2         ; строб Y2, активный 0
PIN_CSEN     BIT P3.4         ; CS_EN: разрешение дешифратора CS (1 — CS активны)

; --- таймеры (кварц 12 МГц) ---
T2_RELOAD_H  EQU 0F7h          ; строб Y2 2152 мкс, Timer1
T2_RELOAD_L  EQU 98h
TICK_H       EQU 0FCh          ; тик 1 мс для Y1/T3, Timer0
TICK_L       EQU 18h
Y1_TICKS     EQU 271           ; T1 = 271 мс
T3_TICKS     EQU 2740          ; T3 = 2740 мс
; ---- конец vars.inc ----
DSEG AT 30h
X1:     DS 1                    ; цифра с клавиатуры (0 — Y2 пассивен)
X2:     DS 1                    ; отсчёт из буфера
CSEG AT 0h
        sjmp START
org 03h ; "заглушка" для INT0
        nop
        reti
org 0bh ; "заглушка" для Tmr0_ovf
        nop
        reti
org 13h ; "заглушка" для INT1
        nop
        reti
org 1bh                         ; Timer1 — конец строба Y2
        clr TR1                    ; остановить отсчёт длительности
        setb PIN_Y2                ; вернуть строб в пассивное состояние
        reti
org 23h ; "заглушка" для UART
        nop
        reti
org 2bh
START:
; Инициализация МК **********************************************
        mov SP, #07h
        setb PIN_CSEN              ; включить дешифратор адреса
        setb PIN_Y2                ; строб неактивен до первого вызова
        mov TMOD, #11h             ; таймеры в режиме 1
        clr TR1                    ; Timer1 остановлен
        setb ET1                   ; разрешить прерывание Timer1
        setb EA                    ; разрешить прерывания глобально
        mov X1, #0                 ; пример: нулевой ввод держит Y2 пассивным
        mov X2, #0
        call Y2Out ; %proc%
        jmp $ ; %stop%

; Y2Out — расчёт Y2 = (G+M+X1+X2) mod 256 (X1 = 0 → Y2 = 0), запись в регистр Y2 и строб T2.
; Вход: X1, X2. Выход: регистр Y2; PIN_Y2 = 0 на T2 мкс (конец — в прерывании Timer1).
; Портит: A, DPTR, флаги PSW, Timer1.
Y2Out:
        mov A, X1                  ; нулевой X1 требует нулевого выхода
        jz Y2Zero
        add A, X2                  ; сложение выполняется по модулю 256
        add A, #Y2_CONST           ; прибавить G + M из vars.inc
        sjmp Y2Write
Y2Zero:
        clr A                      ; пассивное значение при X1 = 0
Y2Write:
        mov DPTR, #ADR_Y2          ; выбрать внешний регистр Y2
        movx @DPTR, A              ; записать значение до начала стробы
        mov TH1, #T2_RELOAD_H      ; загрузить длительность T2
        mov TL1, #T2_RELOAD_L
        clr TF1                    ; сбросить старое переполнение
        clr PIN_Y2                 ; активная часть стробы начинается
        setb TR1                   ; Timer1 завершит строб по прерыванию
        ret
END
