; Сапелин А.С., А-12-23, 18, v1
; ---- vars.inc (вклеен mpscode) ----
; Параметры варианта А-12, M=18 (ТЗ 2026). Сгенерировано mpsgen — не править руками.

VAR_M        EQU 18            ; номер варианта
VAR_G        EQU 12            ; номер группы
Y2_CONST     EQU 30            ; G+M: Y2 = (Y2_CONST + X1 + X2) mod 256

; --- адреса внешних устройств (movx) ---
ADR_IND      EQU 0E000h        ; индикатор, CS Y7
ADR_KB       EQU 2000h        ; клавиатура: запись столбцов / чтение строк, CS Y1
ADR_Y2       EQU 0C000h        ; регистр Y2, CS Y6
BUF_START    EQU 0A000h        ; первая ячейка буфера, CS Y5
BUF_END      EQU 0A6F8h        ; первая ячейка ЗА буфером (точка заворота)
BUF_SIZE     EQU 1784

; --- клавиатура 4x3 и индикатор ---
KB_COLS      EQU 4             ; столбцы Col1…Col4 — биты D0…D3 при записи в ADR_KB (0 — столбец опрашивается)
KB_ROWS      EQU 3             ; строки Row1…Row3 — биты D0…D2 при чтении ADR_KB (0 — замкнута); старшие биты не определены — маскировать
IND_OFF      EQU 0FFh           ; индикатор погашен (общий анод: 0 зажигает сегмент; D0…D7 = a…g, dp)

; --- внутреннее ОЗУ ---
HEAD_L       EQU 47h          ; «голова», младший байт
HEAD_H       EQU 48h          ; «голова», старший байт
TAIL_L       EQU 49h          ; «хвост», младший байт
TAIL_H       EQU 4Ah          ; «хвост», старший байт
F_EMPTY      BIT 12h          ; флаг пустоты буфера (22h.2)
F_OVF        BIT 1Eh          ; флаг переполнения буфера (23h.6)

; --- выводы ---
PIN_Y1       BIT P1.4         ; строб Y1, активный 0
PIN_Y2       BIT P1.5         ; строб Y2, активный 0
PIN_CSEN     BIT P3.4         ; CS_EN: разрешение дешифратора CS (1 — CS активны)

; --- таймеры (кварц 12 МГц) ---
T2_RELOAD_H  EQU 0F6h          ; строб Y2 2452 мкс, Timer1
T2_RELOAD_L  EQU 6Ch
TICK_H       EQU 0FCh          ; тик 1 мс для Y1/T3, Timer0
TICK_L       EQU 18h
Y1_TICKS     EQU 371           ; T1 = 371 мс
T3_TICKS     EQU 3240          ; T3 = 3240 мс
; ---- конец vars.inc ----
DSEG AT 30h
CSEG AT 0h
        sjmp START
org 03h
        nop
        reti
org 0bh
        nop
        reti
org 13h
        nop
        reti
org 1bh
        nop
        reti
org 23h
        nop
        reti
org 2bh
START:
        mov SP, #07h            ; начальный стек по шаблону ТЗ
        mov IE, #0              ; самостоятельная программа не обслуживает IRQ
        setb PIN_CSEN           ; доступ к ОЗУ разрешается дешифратором
        mov HEAD_L, #LOW BUF_START ; следующая запись начнётся с начала кольца
        mov HEAD_H, #HIGH BUF_START
        mov TAIL_L, #LOW BUF_START ; чтение и запись пока имеют общий адрес
        mov TAIL_H, #HIGH BUF_START
        setb F_EMPTY            ; в памяти ещё нет действительных отсчётов
        clr F_OVF               ; исходное состояние не является переполнением
        call BufRead ; %proc%
        jmp $ ; %stop%

; Назначение: BufRead извлекает самый старый отсчёт из кольцевого буфера.
; Вход: HEAD, TAIL и флаги из vars.inc; данные во внешнем ОЗУ.
; Выход: A — прочитанный байт, при пустом буфере A = 0 без изменения указателей и флагов.
; Выход: после чтения TAIL передвинут, F_OVF = 0, F_EMPTY отражает опустошение.
; Портит: A, DPTR, флаги PSW; B сохраняется.
BufRead:
        jnb F_EMPTY, ReadData    ; пустой буфер нельзя читать и передвигать
        clr A
        ret
ReadData:
        push B                  ; временное хранение отсчёта не меняет B вызывающей программы
        mov DPL, TAIL_L
        mov DPH, TAIL_H          ; адрес самого старого отсчёта
        movx A, @DPTR
        mov B, A                ; сохранить результат на время проверки границ
        inc DPTR                ; предварительно перейти к следующему отсчёту
        mov A, DPL
        cjne A, #LOW BUF_END, ReadStore
        mov A, DPH
        cjne A, #HIGH BUF_END, ReadStore
        mov DPTR, #BUF_START     ; конец кольца переходит в его начало
ReadStore:
        mov TAIL_L, DPL
        mov TAIL_H, DPH          ; зафиксировать новый адрес чтения
        clr F_OVF               ; прочитанный отсчёт освободил место
        clr F_EMPTY
        mov A, DPL
        cjne A, HEAD_L, ReadDone ; разные указатели означают оставшиеся данные
        mov A, DPH
        cjne A, HEAD_H, ReadDone
        setb F_EMPTY            ; хвост догнал голову после последнего отсчёта
ReadDone:
        mov A, B                ; вернуть именно отсчёт, а не байт адреса
        pop B
        ret
END
