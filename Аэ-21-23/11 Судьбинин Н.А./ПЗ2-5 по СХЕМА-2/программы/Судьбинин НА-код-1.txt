; Судьбинин Н.А., Аэ-21-23, 11, v1
; ---- vars.inc (вклеен mpscode) ----
; Параметры варианта Аэ-21, M=11 (ТЗ 2026). Сгенерировано mpsgen — не править руками.

VAR_M        EQU 11            ; номер варианта
VAR_G        EQU 21            ; номер группы
Y2_CONST     EQU 32            ; G+M: Y2 = (Y2_CONST + X1 + X2) mod 256

; --- адреса внешних устройств (movx) ---
ADR_IND      EQU 2000h        ; индикатор, CS Y1
ADR_KB       EQU 6000h        ; клавиатура: запись столбцов / чтение строк, CS Y3
ADR_Y2       EQU 4000h        ; регистр Y2, CS Y2
BUF_START    EQU 0E000h        ; первая ячейка буфера, CS Y7
BUF_END      EQU 0E776h        ; первая ячейка ЗА буфером (точка заворота)
BUF_SIZE     EQU 1910
BUF_X2IN     EQU 0FFFFh        ; ячейка обмена: сюда внешнее устройство пишет X2 правым портом (строб X2 = прерывание INT1)

; --- клавиатура 3x4 и индикатор ---
KB_COLS      EQU 3             ; столбцы Col1…Col3 — биты D0…D2 при записи в ADR_KB (0 — столбец опрашивается)
KB_ROWS      EQU 4             ; строки Row1…Row4 — биты D0…D3 при чтении ADR_KB (0 — замкнута); старшие биты не определены — маскировать
IND_OFF      EQU 00h           ; индикатор погашен (общий катод: 1 зажигает сегмент; D0…D7 = a…g, dp)

; --- внутреннее ОЗУ ---
HEAD_L       EQU 51h          ; «голова», младший байт
HEAD_H       EQU 52h          ; «голова», старший байт
TAIL_L       EQU 53h          ; «хвост», младший байт
TAIL_H       EQU 54h          ; «хвост», старший байт
F_EMPTY      BIT 0Bh          ; флаг пустоты буфера (21h.3)
F_OVF        BIT 20h          ; флаг переполнения буфера (24h.0)

; --- выводы ---
PIN_Y1       BIT P1.4         ; строб Y1, активный 0
PIN_Y2       BIT P1.5         ; строб Y2, активный 0
PIN_CSEN     BIT P3.4         ; CS_EN: разрешение дешифратора CS (1 — CS активны)

; --- таймеры (кварц 12 МГц) ---
T2_RELOAD_H  EQU 0F9h          ; строб Y2 1751 мкс, Timer1
T2_RELOAD_L  EQU 29h
TICK_H       EQU 0FCh          ; тик 1 мс для Y1/T3, Timer0
TICK_L       EQU 18h
Y1_TICKS     EQU 313           ; T1 = 313 мс
T3_TICKS     EQU 2745          ; T3 = 2745 мс
; ---- конец vars.inc ----
DSEG AT 30h
; переменных в ОЗУ нет: всё держится в регистрах банка 0 и в стеке
CSEG AT 0h
        sjmp START
org 03h                         ; INT0 — нажатие клавиши: процедура вызывается отсюда
        call KbScan ; %proc%
        reti
org 0bh ; "заглушка" для Tmr0_ovf
        nop
        reti
org 13h ; "заглушка" для INT1
        nop
        reti
org 1bh ; "заглушка" для Tmr1_ovf
        nop
        reti
org 23h ; "заглушка" для UART
        nop
        reti
org 2bh
START:
; Инициализация МК **********************************************
        mov SP, #07h            ; стек сразу над банком 0
        setb PIN_CSEN           ; разрешаем дешифратор адреса, иначе клавиатура не выбирается
        mov DPTR, #ADR_KB
        clr A
        movx @DPTR, A           ; опускаем все столбцы: любое нажатие замкнёт строку на 0 и даст INT0
        setb IT0                ; INT0 ловим по спаду — одно нажатие = один вызов
        setb EX0                ; разрешаем прерывание от клавиатуры
        setb EA                 ; и прерывания вообще
        jmp $ ; %stop%

; KbScan — опрос матрицы клавиатуры 3x4 (вызывается из обработчика INT0).
; Столбцы опускаются в 0 по одному, по прочитанным строкам ищется единственная нажатая клавиша.
; Вход: нет. Выход: A — код клавиши по табл. 3 ТЗ (0…9, «Т» = 10, «С» = 11) или 0FFh,
; если не нажато ничего либо нажато две клавиши и больше. После опроса все столбцы снова в 0.
; Портит только A: PSW, DPTR, B, R2–R5 сохраняются в стеке.
KbScan:
        push PSW
        push B
        push DPL
        push DPH
        push AR2
        push AR3
        push AR4
        push AR5
        mov DPTR, #ADR_KB
        mov R4, #0FFh           ; R4 — найденный код; 0FFh — пока ничего не нашли
        mov R2, #0              ; R2 — номер текущего столбца (0 = Col1)
        mov R3, #11111110b      ; R3 — что пишем в столбцы: 0 только у опрашиваемого
KsCol:
        mov A, R3
        movx @DPTR, A           ; опускаем один столбец, остальные держим в 1
        movx A, @DPTR           ; читаем, какие строки он притянул к 0
        anl A, #(1 SHL KB_ROWS) - 1   ; оставляем только подключённые строки Row1…Row4
        xrl A, #(1 SHL KB_ROWS) - 1   ; переворачиваем: теперь 1 = клавиша на этой строке нажата
        jz KsNext               ; в этом столбце ничего — к следующему
        cjne R4, #0FFh, KsBad   ; клавиша уже была в другом столбце — нажато две
        mov B, A
        dec A
        anl A, B                ; x AND (x-1) = 0 только если нажата ровно одна строка
        jnz KsBad               ; две строки в одном столбце — ошибка
        mov A, B
        mov R5, #0              ; R5 — номер строки, ищем единственную единицу
KsRow:
        rrc A                   ; выталкиваем младшую строку в перенос
        jc KsHit                ; нашли нажатую строку
        inc R5
        sjmp KsRow
KsHit:
        mov A, R5
        mov B, #KB_COLS
        mul AB                  ; позиция клавиши в таблице = строка * 3 + столбец
        add A, R2
        mov DPTR, #KbCodes
        movc A, @A+DPTR         ; берём код клавиши по раскладке рис. 4 ТЗ
        mov R4, A               ; запоминаем, но опрос продолжаем — вдруг нажата ещё одна
        mov DPTR, #ADR_KB       ; возвращаем адрес клавиатуры
KsNext:
        mov A, R3
        rl A                    ; ноль переезжает на следующий столбец
        mov R3, A
        inc R2
        cjne R2, #KB_COLS, KsCol      ; пока не прошли все три столбца
        mov A, R4               ; итог: код единственной клавиши или 0FFh (ничего не нажато)
        sjmp KsEnd
KsBad:
        mov A, #0FFh            ; нажато больше одной клавиши
KsEnd:
        mov R4, A               ; придерживаем результат
        mov DPTR, #ADR_KB
        clr A
        movx @DPTR, A           ; все столбцы обратно в 0 — следующее нажатие снова даст INT0
        clr IE0                 ; пока перебирали столбцы, линия INT0 дёргалась — гасим ложный запрос
        mov A, R4
        pop AR5
        pop AR4
        pop AR3
        pop AR2
        pop DPH
        pop DPL
        pop B
        pop PSW
        ret

; Коды клавиш по строкам (Row1…Row4), в строке — Col1…Col3 (рис. 4 ТЗ, раскладка 3x4)
KbCodes:
        DB 1, 2, 3
        DB 4, 5, 6
        DB 7, 8, 9
        DB 0, 10, 11            ; «0», «Т», «С»
END
